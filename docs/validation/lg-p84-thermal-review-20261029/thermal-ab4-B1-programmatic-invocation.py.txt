class SupplyMatchedAdb(instant.InstantAdb):
    def snapshot(self):
        row=super().snapshot()

def service_state(text):
    fields={}
    for key,pattern in {'AC':r'AC powered:\s*(true|false)','USB':r'USB powered:\s*(true|false)','wireless':r'Wireless powered:\s*(true|false)','status':r'^\s*status:\s*(\d+)'}.items():
        match=re.search(pattern,text,re.M)
        if match:fields[key]=match.group(1)
    return fields
